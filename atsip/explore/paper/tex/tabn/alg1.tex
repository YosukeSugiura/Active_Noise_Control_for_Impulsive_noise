\begin{algorithm}[tbp]
\caption{Processing of the proposed algorithm in block $n$}
\label{alg:block}
\begin{algorithmic}[1]
\Require $\mathbf{w}_{\mathrm f}$, the state of the trial ($n_0$, $\mathbf{w}_{\mathrm b}$), $\hat{\mathbf{S}}_{rr,n-1}$, $\bar{z}_{n-1}$
\State Compute $\mathbf{R}_n$, $\hat{\mathbf{S}}_{rr,n}$, $\mathbf{F}_n$, and $g_n$ by \eqref{eq:5} and \eqref{eq:6}, and $\hat{d}(k)$ by \eqref{eq:7}
\State $T \gets$ \textbf{true} if a trial is active, and $\mathit{moved} \gets$ \textbf{false}
\If{$T$} \Comment{$m = n - n_0 \ge 1$}
  \If{$m \ge 2$}
    \State Reconstruct $\tilde{e}_{\mathrm b}$ and $\tilde{e}_{\mathrm f}$ by \eqref{eq:8} and compute $\rho_n$ by \eqref{eq:11}
    \If{$\rho_n > \rho_{\mathrm{abort}}$ or $\rho_n$ is not finite} discard $\mathbf{w}_{\mathrm b}$
    \ElsIf{$m \ge n_1$ and $\rho_n \le \rho_1$} $\mathbf{w}_{\mathrm f} \gets \mathbf{w}_{\mathrm b}$, $\mathit{moved} \gets$ \textbf{true}
    \ElsIf{$m = n_2$ and $\rho_n \le \rho_2$} $\mathbf{w}_{\mathrm f} \gets \mathbf{w}_{\mathrm b}$, $\mathit{moved} \gets$ \textbf{true}
    \ElsIf{$m = n_2$} discard $\mathbf{w}_{\mathrm b}$
    \EndIf
  \EndIf
  \If{the trial is not ended} update $\mathbf{w}_{\mathrm b}$ by \eqref{eq:4} with $\sigma_{\mathrm E}$ and the frame \eqref{eq:8} of $\mathbf{w}_{\mathrm b}$
  \EndIf
\EndIf
\If{$\mathit{moved}$}
  \State Keep $\mathbf{w}_{\mathrm f}$ in the early phase until block $n_0 + n_{\mathrm E}$; no foreground update in this block
\Else
  \State Update $\mathbf{w}_{\mathrm f}$ by \eqref{eq:4} in its phase: frame \eqref{eq:8} of $\mathbf{w}_{\mathrm f}$ in the early phase, measured frame in the late phase
\EndIf
\State Compute $z_n$ by \eqref{eq:10} and update $\bar{z}_n$ unless it is held or $z_n$ is not finite
\If{$z_n > \gamma \bar{z}_{n-1}$ for two consecutive blocks, not $T$, and no trial started within $n_{\mathrm G}$ blocks}
  \State Start a trial: $n_0 \gets n$, $\mathbf{w}_{\mathrm b} \gets \mathbf{w}_{\mathrm f}$ (the coefficients before the update of this block)
\EndIf
\State The control signal of the next block is generated by $\mathbf{w}_{\mathrm f}$ only
\end{algorithmic}
\end{algorithm}
